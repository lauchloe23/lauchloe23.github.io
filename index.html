public boolean isValidPlay(String strCardName){
    if(strCardName.startsWith("wild")) return true;
    if(intDiscardPileSize == 0) return true;
    String strTopCard = strDiscardPile[intDiscardPileSize - 1][0];

    // after a wild, only the colour the player picked is legal
    if(strTopCard.startsWith("wild")){
        if(!strWildColor.equals("")){
            return getColor(strCardName).equals(strWildColor);
        }
        return true;
    }
    return getColor(strCardName).equals(getColor(strTopCard))
        || getValue(strCardName).equals(getValue(strTopCard));
}
